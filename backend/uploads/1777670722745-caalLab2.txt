.data
#data slectionincludes string to be output to the console
msg1: .asciiz"Enter the first number: "
msg2: .asciiz"\n Enter the second number: "
msg3: .asciiz"\n The result of addition is:"
.text
.globl main
.ent main
main:
li $v0,4  #prep for string display
la $a0,msg1  #prep for string display
syscall

li,v0,5   
syscall
move $t1,$v0 
li $v0, 4
la $a0,msg2
syscall
li $v0,5
syscall
move $t2,$v0
add $t3, $t1, $t2
li $v0,4
la $a0, msg3
syscall
li $v0,1
move $a0, $t3
syscall

jr $ra
.end main
